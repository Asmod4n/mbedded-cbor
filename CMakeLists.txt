cmake_minimum_required(VERSION 3.25)
project(mbedded-cbor VERSION 0.9.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

set(CBOR_COMPILE_DEFINITIONS _GLIBCXX_ASSERTIONS _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)
add_library(mbedded-cbor INTERFACE)
add_library(mbedded-cbor::mbedded-cbor ALIAS mbedded-cbor)
target_include_directories(mbedded-cbor INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(mbedded-cbor INTERFACE cxx_std_23)
target_compile_definitions(mbedded-cbor INTERFACE ${CBOR_COMPILE_DEFINITIONS})

option(CBOR_INSTALL "Install the headers, the CMake package and the pkg-config file" ${PROJECT_IS_TOP_LEVEL})
if(CBOR_INSTALL)
	set(CBOR_CMAKE_DIR ${CMAKE_INSTALL_DATADIR}/cmake/mbedded-cbor)
	install(TARGETS mbedded-cbor EXPORT mbedded-cbor-targets)
	install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
	install(EXPORT mbedded-cbor-targets NAMESPACE mbedded-cbor:: DESTINATION ${CBOR_CMAKE_DIR})
	configure_package_config_file(cmake/mbedded-cbor-config.cmake.in mbedded-cbor-config.cmake INSTALL_DESTINATION ${CBOR_CMAKE_DIR})
	write_basic_package_version_file(mbedded-cbor-config-version.cmake COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
	install(FILES ${PROJECT_BINARY_DIR}/mbedded-cbor-config.cmake ${PROJECT_BINARY_DIR}/mbedded-cbor-config-version.cmake DESTINATION ${CBOR_CMAKE_DIR})
	cmake_path(RELATIVE_PATH CMAKE_INSTALL_PREFIX BASE_DIRECTORY ${CMAKE_INSTALL_FULL_DATADIR}/pkgconfig OUTPUT_VARIABLE CBOR_PC_PREFIX)
	string(REGEX REPLACE "/$" "" CBOR_PC_PREFIX ${CBOR_PC_PREFIX})
	list(TRANSFORM CBOR_COMPILE_DEFINITIONS PREPEND -D OUTPUT_VARIABLE CBOR_PC_DEFINITIONS)
	list(JOIN CBOR_PC_DEFINITIONS " " CBOR_PC_DEFINITIONS)
	configure_file(cmake/mbedded-cbor.pc.in mbedded-cbor.pc @ONLY)
	install(FILES ${PROJECT_BINARY_DIR}/mbedded-cbor.pc DESTINATION ${CMAKE_INSTALL_DATADIR}/pkgconfig)
endif()

set(CBOR_WARNINGS -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion -Wold-style-cast $<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:-Wnull-dereference> -Werror)

if(PROJECT_IS_TOP_LEVEL)
	include(CTest)
endif()
if(PROJECT_IS_TOP_LEVEL AND BUILD_TESTING)
	find_package(doctest REQUIRED)
	include(doctest)

	if(CMAKE_CXX_FLAGS MATCHES "-fsanitize=[a-z,]*thread")
		get_target_property(CBOR_TEST_SOURCES mbedded-cbor-test SOURCES)
		list(REMOVE_ITEM CBOR_TEST_SOURCES test/allocation.cpp)
		set_target_properties(mbedded-cbor-test PROPERTIES SOURCES "${CBOR_TEST_SOURCES}")

	set_source_files_properties(test/schema.cpp PROPERTIES UNITY_GROUP schema)

		target_compile_features(mbedded-cbor-test PRIVATE cxx_std_26)

option(CBOR_FUZZ "Build one fuzz target per public API: libFuzzer with clang, or AFL++ with CBOR_FUZZ_DRIVER" OFF)
set(CBOR_FUZZ_DRIVER "" CACHE FILEPATH "libAFLDriver.a of AFL++; empty for libFuzzer")
if(CBOR_FUZZ)
	find_package(doctest REQUIRED)
	include(CheckCXXSourceCompiles)
	set(CMAKE_REQUIRED_FLAGS "-std=c++26 -freflection")
	check_cxx_source_compiles("#if !__cpp_impl_reflection\n#error\n#endif\nint main() {}" CBOR_FUZZ_HAS_FREFLECTION)
	unset(CMAKE_REQUIRED_FLAGS)
	set(CBOR_FUZZ_TARGETS decode encode encoder item_size lazy path)
	if(CBOR_FUZZ_HAS_FREFLECTION)
		list(APPEND CBOR_FUZZ_TARGETS schema_decode schema_encode)
	endif()
	foreach(target IN LISTS CBOR_FUZZ_TARGETS)
		if(target MATCHES "^schema_(.*)$")
			set(function "fuzz::schema::${CMAKE_MATCH_1}_target")
		else()
			set(function "fuzz::${target}_target")
		endif()
		add_executable(fuzz-${target} fuzz/target.cpp)
		target_link_libraries(fuzz-${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(fuzz-${target} PRIVATE ${CBOR_WARNINGS})
		target_compile_definitions(fuzz-${target} PRIVATE DOCTEST_CONFIG_DISABLE FUZZ_TARGET=${function})
		if(CBOR_FUZZ_HAS_FREFLECTION)
			target_compile_features(fuzz-${target} PRIVATE cxx_std_26)
			target_compile_options(fuzz-${target} PRIVATE -freflection)
		endif()
		if(CBOR_FUZZ_DRIVER)
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_libraries(fuzz-${target} PRIVATE ${CBOR_FUZZ_DRIVER})
		else()
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
		endif()
	endforeach()
endif()
