	function(cbor_test_settings target)
		cmake_parse_arguments(PARSE_ARGV 1 arg "" "PCH_FROM" "PCH_HEADERS")
		if(NOT arg_PCH_HEADERS)
			set(arg_PCH_HEADERS ${CBOR_TEST_FULL_PCH_HEADERS})
		endif()
		target_link_libraries(${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(${target} PRIVATE ${CBOR_WARNINGS} $<$<BOOL:${CBOR_HAS_WC2Y_EXTENSIONS}>:-Wno-c2y-extensions>)
		if(CBOR_HAS_FREFLECTION)
			target_compile_features(${target} PRIVATE cxx_std_26)
			target_compile_options(${target} PRIVATE -freflection)
		endif()
		target_compile_definitions(${target} PRIVATE DOCTEST_COUNTER=__COUNTER__)
		foreach(macro directory IN ZIP_LISTS CBOR_TEST_DATA_MACROS CBOR_TEST_DATA_DIRECTORIES)
			target_compile_definitions(${target} PRIVATE ${macro}="${directory}")
		endforeach()
		if(arg_PCH_FROM STREQUAL "${target}")
			target_precompile_headers(${target} PRIVATE ${arg_PCH_HEADERS})
		elseif(NOT arg_PCH_FROM STREQUAL "none")
			target_precompile_headers(${target} REUSE_FROM ${arg_PCH_FROM})
		endif()
	endfunction()

option(CBOR_FUZZ "Build one fuzz target per public API: libFuzzer with clang, or AFL++ with CBOR_FUZZ_DRIVER" OFF)
set(CBOR_FUZZ_DRIVER "" CACHE FILEPATH "libAFLDriver.a of AFL++; empty for libFuzzer")
option(CBOR_FUZZ_SMALL_BUFFERS "Build the fuzz targets with the smallest block of encode_iov and zone chunk, so short inputs cross every boundary" OFF)
option(CBOR_FUZZ_GLIBCXX_DEBUG "Build the fuzz targets with the debug mode of libstdc++, without -freflection, on the C++23 path" OFF)
if(CBOR_FUZZ)
	find_package(doctest REQUIRED)
	enable_testing()
	set(CBOR_FUZZ_ENVIRONMENT
		ASAN_OPTIONS=abort_on_error=1:halt_on_error=1:symbolize=0:detect_stack_use_after_return=1:max_malloc_fill_size=1073741824:quarantine_size_mb=16
		UBSAN_OPTIONS=halt_on_error=1:abort_on_error=1:print_stacktrace=1)
	file(GLOB CBOR_FUZZ_CORPUS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/fuzz/corpus/*)
	set(CBOR_FUZZ_REFLECTION OFF)
	if(CBOR_FUZZ_GLIBCXX_DEBUG)
		message(STATUS "mbedded-cbor: CBOR_FUZZ_GLIBCXX_DEBUG builds the fuzz targets without -freflection, because the debug mode of libstdc++ does not compile with it; the targets schema_decode and schema_encode need reflection and are not built")
	else()
		include(CheckCXXSourceCompiles)
		set(CMAKE_REQUIRED_FLAGS "-std=c++26 -freflection")
		check_cxx_source_compiles("#if !__cpp_impl_reflection\n#error\n#endif\nint main() {}" CBOR_FUZZ_HAS_FREFLECTION)
		unset(CMAKE_REQUIRED_FLAGS)
		set(CBOR_FUZZ_REFLECTION ${CBOR_FUZZ_HAS_FREFLECTION})
	endif()
	set(CBOR_FUZZ_TARGETS decode diagnostic item_size lazy path sequence_reader structs_decode)
	if(CBOR_FUZZ_REFLECTION)
		list(APPEND CBOR_FUZZ_TARGETS schema_decode schema_encode)
	endif()
	foreach(target IN LISTS CBOR_FUZZ_TARGETS)
		if(target MATCHES "^(schema|structs)_(.*)$")
			set(function "fuzz::${CMAKE_MATCH_1}::${CMAKE_MATCH_2}_target")
			set(header "${CMAKE_MATCH_1}.hpp")
		else()
			set(function "fuzz::${target}_target")
			set(header "${target}.hpp")
		endif()
		add_executable(fuzz-${target} fuzz/target.cpp)
		target_link_libraries(fuzz-${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(fuzz-${target} PRIVATE ${CBOR_WARNINGS})
		target_compile_definitions(fuzz-${target} PRIVATE DOCTEST_CONFIG_DISABLE FUZZ_TARGET=${function} "FUZZ_HEADER=\"${header}\"" CBOR_ENCODER_COPYBREAK=0)
		if(CBOR_FUZZ_REFLECTION)
			target_compile_features(fuzz-${target} PRIVATE cxx_std_26)
			target_compile_options(fuzz-${target} PRIVATE -freflection)
		endif()
		if(CBOR_FUZZ_GLIBCXX_DEBUG)
			target_compile_definitions(fuzz-${target} PRIVATE _GLIBCXX_DEBUG _GLIBCXX_DEBUG_PEDANTIC)
		endif()
		if(CBOR_FUZZ_SMALL_BUFFERS)
			target_compile_definitions(fuzz-${target} PRIVATE CBOR_ENCODER_BLOCK_SIZE=9 CBOR_ENCODER_INLINE_BUFFER_SIZE=9 CBOR_STRING_SAVER_CHUNK_SIZE=16)
		endif()
		if(CBOR_FUZZ_DRIVER)
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_libraries(fuzz-${target} PRIVATE ${CBOR_FUZZ_DRIVER})
			add_test(NAME fuzz_replay_${target} COMMAND fuzz-${target} ${CBOR_FUZZ_CORPUS})
		else()
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
			set(artifacts ${PROJECT_BINARY_DIR}/fuzz-artifacts/${target})
			file(MAKE_DIRECTORY ${artifacts})
			add_test(NAME fuzz_replay_${target} COMMAND fuzz-${target} -runs=0 -artifact_prefix=${artifacts}/ ${PROJECT_SOURCE_DIR}/fuzz/corpus)
		endif()
		set_tests_properties(fuzz_replay_${target} PROPERTIES ENVIRONMENT "${CBOR_FUZZ_ENVIRONMENT}" WORKING_DIRECTORY ${PROJECT_BINARY_DIR})
	endforeach()
endif()
