cmake_minimum_required(VERSION 3.25)
project(mbedded-cbor LANGUAGES CXX)

add_library(mbedded-cbor INTERFACE)
target_include_directories(mbedded-cbor INTERFACE include)
target_compile_features(mbedded-cbor INTERFACE cxx_std_23)
target_compile_definitions(mbedded-cbor INTERFACE _GLIBCXX_ASSERTIONS _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_FAST)

set(CBOR_WARNINGS -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion -Wold-style-cast $<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:-Wnull-dereference> -Werror)

include(CTest)
if(BUILD_TESTING)
	find_package(doctest REQUIRED)
	include(doctest)
	add_executable(mbedded-cbor-test test/main.cpp test/schema.cpp test/value.cpp test/sequence.cpp test/diagnostic_notation.cpp test/fixed_width.cpp test/error.cpp test/path.cpp test/doc_path.cpp test/doc_path.cpp test/allocation.cpp test/self_described.cpp test/validity.cpp test/bignum.cpp test/item_size.cpp test/lazy.cpp test/nesting_depth_limit.cpp test/cts.cpp test/limits.cpp test/string.cpp test/head.cpp test/databind.cpp test/shared.cpp test/vectors.cpp test/simple_value.cpp test/item.cpp test/float.cpp)
	if(CMAKE_CXX_FLAGS MATCHES "-fsanitize=[a-z,]*thread")
		get_target_property(CBOR_TEST_SOURCES mbedded-cbor-test SOURCES)
		list(REMOVE_ITEM CBOR_TEST_SOURCES test/allocation.cpp)
		set_target_properties(mbedded-cbor-test PROPERTIES SOURCES "${CBOR_TEST_SOURCES}")
	endif()
	target_link_libraries(mbedded-cbor-test PRIVATE mbedded-cbor doctest::doctest)
	target_compile_options(mbedded-cbor-test PRIVATE ${CBOR_WARNINGS} $<$<CXX_COMPILER_ID:Clang>:-Wno-c2y-extensions>)
	target_precompile_headers(mbedded-cbor-test PRIVATE <cbor/cbor.hpp> <doctest/doctest.h>)
	set_source_files_properties(test/main.cpp PROPERTIES SKIP_PRECOMPILE_HEADERS ON SKIP_UNITY_BUILD_INCLUSION ON)
	set_target_properties(mbedded-cbor-test PROPERTIES UNITY_BUILD ON UNITY_BUILD_MODE GROUP)
	set_source_files_properties(test/schema.cpp PROPERTIES UNITY_GROUP schema)
	set_source_files_properties(test/path.cpp test/doc_path.cpp test/allocation.cpp test/self_described.cpp test/value.cpp test/string.cpp test/diagnostic_notation.cpp test/item_size.cpp PROPERTIES UNITY_GROUP batch_0)
	set_source_files_properties(test/lazy.cpp test/nesting_depth_limit.cpp test/cts.cpp test/limits.cpp test/validity.cpp test/item.cpp test/head.cpp test/fixed_width.cpp PROPERTIES UNITY_GROUP batch_1)
	set_source_files_properties(test/databind.cpp test/shared.cpp test/vectors.cpp test/simple_value.cpp test/sequence.cpp test/bignum.cpp test/float.cpp test/error.cpp PROPERTIES UNITY_GROUP batch_2)
	include(CheckCXXCompilerFlag)
	check_cxx_compiler_flag("-std=c++26 -freflection -Werror" CBOR_HAS_FREFLECTION)
	if(CBOR_HAS_FREFLECTION)
		target_compile_features(mbedded-cbor-test PRIVATE cxx_std_26)
		target_compile_options(mbedded-cbor-test PRIVATE -freflection)
	endif()
	target_compile_definitions(mbedded-cbor-test PRIVATE DOCTEST_COUNTER=__COUNTER__ TEST_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/test-vectors" CBOR_TEST_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/cbor-test-vectors" JSONPATH_CTS="${CMAKE_CURRENT_SOURCE_DIR}/test/jsonpath")
	doctest_discover_tests(mbedded-cbor-test)
endif()

option(CBOR_FUZZ "Build one fuzz target per public API: libFuzzer with clang, or AFL++ with CBOR_FUZZ_DRIVER" OFF)
set(CBOR_FUZZ_DRIVER "" CACHE FILEPATH "libAFLDriver.a of AFL++; empty for libFuzzer")
if(CBOR_FUZZ)
	find_package(doctest REQUIRED)
	include(CheckCXXSourceCompiles)
	set(CMAKE_REQUIRED_FLAGS "-std=c++26 -freflection")
	check_cxx_source_compiles("#if !__cpp_impl_reflection\n#error\n#endif\nint main() {}" CBOR_FUZZ_HAS_FREFLECTION)
	unset(CMAKE_REQUIRED_FLAGS)
	set(CBOR_FUZZ_TARGETS decode encode encoder item_size lazy path)
	if(CBOR_FUZZ_HAS_FREFLECTION)
		list(APPEND CBOR_FUZZ_TARGETS schema_decode schema_encode)
	endif()
	foreach(target IN LISTS CBOR_FUZZ_TARGETS)
		if(target MATCHES "^schema_(.*)$")
			set(function "fuzz::schema::${CMAKE_MATCH_1}_target")
		else()
			set(function "fuzz::${target}_target")
		endif()
		add_executable(fuzz-${target} fuzz/target.cpp)
		target_link_libraries(fuzz-${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(fuzz-${target} PRIVATE ${CBOR_WARNINGS})
		target_compile_definitions(fuzz-${target} PRIVATE DOCTEST_CONFIG_DISABLE FUZZ_TARGET=${function})
		if(CBOR_FUZZ_HAS_FREFLECTION)
			target_compile_features(fuzz-${target} PRIVATE cxx_std_26)
			target_compile_options(fuzz-${target} PRIVATE -freflection)
		endif()
		if(CBOR_FUZZ_DRIVER)
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)
			target_link_libraries(fuzz-${target} PRIVATE ${CBOR_FUZZ_DRIVER})
		else()
			target_compile_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
			target_link_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
		endif()
	endforeach()
endif()
