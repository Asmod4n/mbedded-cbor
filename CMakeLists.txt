cmake_minimum_required(VERSION 3.25)
project(mbedded-cbor LANGUAGES CXX)

add_library(mbedded-cbor INTERFACE)
target_include_directories(mbedded-cbor INTERFACE include)
target_compile_features(mbedded-cbor INTERFACE cxx_std_23)

find_package(simdutf QUIET)
if(simdutf_FOUND)
	target_link_libraries(mbedded-cbor INTERFACE simdutf::simdutf)
	target_compile_definitions(mbedded-cbor INTERFACE CBOR_SIMDUTF=1)
endif()

include(CTest)
if(BUILD_TESTING)
	find_package(doctest REQUIRED)
	include(doctest)
	add_executable(mbedded-cbor-test test/main.cpp test/float.cpp test/head.cpp test/string.cpp test/value.cpp test/shared.cpp test/error.cpp test/bignum.cpp test/doc_end.cpp test/lazy.cpp test/path.cpp test/vectors.cpp test/corpus.cpp test/fixed_width.cpp test/schema.cpp)
	target_link_libraries(mbedded-cbor-test PRIVATE mbedded-cbor doctest::doctest)
	include(CheckCXXCompilerFlag)
	check_cxx_compiler_flag("-std=c++26 -freflection" CBOR_HAS_FREFLECTION)
	if(CBOR_HAS_FREFLECTION)
		target_compile_features(mbedded-cbor-test PRIVATE cxx_std_26)
		target_compile_options(mbedded-cbor-test PRIVATE -freflection)
	endif()
	include(CheckCXXCompilerFlag)
	check_cxx_compiler_flag("-std=c++26 -freflection" CBOR_HAS_FREFLECTION)
	if(CBOR_HAS_FREFLECTION)
		target_compile_features(mbedded-cbor-test PRIVATE cxx_std_26)
		target_compile_options(mbedded-cbor-test PRIVATE -freflection)
	endif()
	target_compile_definitions(mbedded-cbor-test PRIVATE TEST_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/test-vectors"
	                                                    FUZZ_CORPUS="${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus")
	doctest_discover_tests(mbedded-cbor-test)
endif()

option(CBOR_FUZZ "Build the libFuzzer target; needs clang" OFF)
if(CBOR_FUZZ)
	find_package(doctest REQUIRED)
	add_executable(mbedded-cbor-fuzzer fuzz/fuzzer.cpp)
	target_link_libraries(mbedded-cbor-fuzzer PRIVATE mbedded-cbor doctest::doctest)
	target_compile_definitions(mbedded-cbor-fuzzer PRIVATE DOCTEST_CONFIG_DISABLE)
	target_compile_options(mbedded-cbor-fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
	target_link_options(mbedded-cbor-fuzzer PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
