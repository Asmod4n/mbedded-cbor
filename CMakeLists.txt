	set(CBOR_BOOST_PFR system)
	target_link_libraries(mbedded-cbor INTERFACE Boost::headers)
	message(STATUS "mbedded-cbor: Boost.PFR ${Boost_VERSION} of the system")
else()
	set(CBOR_BOOST_PFR bundled)
	set(CBOR_BOOST_PFR_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost_pfr/include)
	target_include_directories(mbedded-cbor SYSTEM INTERFACE $<BUILD_INTERFACE:${CBOR_BOOST_PFR_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/mbedded-cbor/boost_pfr>)
	message(STATUS "mbedded-cbor: Boost.PFR 1.92.0 of third_party/boost_pfr, because the system has no Boost 1.84 or later")
endif()

if(CBOR_FUZZ)
	find_package(doctest REQUIRED)
	include(CheckCXXSourceCompiles)
	set(CMAKE_REQUIRED_FLAGS "-std=c++26 -freflection")
	check_cxx_source_compiles("#if !__cpp_impl_reflection\n#error\n#endif\nint main() {}" CBOR_FUZZ_HAS_FREFLECTION)
	unset(CMAKE_REQUIRED_FLAGS)
	set(CBOR_FUZZ_TARGETS decode diagnostic encode encoder item_size lazy path sequence_reader structs_decode)

		list(APPEND CBOR_FUZZ_TARGETS schema_decode schema_encode)
	endif()
	foreach(target IN LISTS CBOR_FUZZ_TARGETS)
		if(target MATCHES "^(schema|structs)_(.*)$")
			set(function "fuzz::${CMAKE_MATCH_1}::${CMAKE_MATCH_2}_target")
		else()
			set(function "fuzz::${target}_target")
		endif()
		add_executable(fuzz-${target} fuzz/target.cpp)
		target_link_libraries(fuzz-${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(fuzz-${target} PRIVATE ${CBOR_WARNINGS})
		target_compile_definitions(fuzz-${target} PRIVATE DOCTEST_CONFIG_DISABLE FUZZ_TARGET=${function})

			target_compile_features(fuzz-${target} PRIVATE cxx_std_26)
			target_compile_options(fuzz-${target} PRIVATE -freflection)
		endif()

			target_compile_options(fuzz-${target} PRIVATE -fsanitize=address,undefined)

		else()

			target_link_options(fuzz-${target} PRIVATE -fsanitize=fuzzer,address,undefined)
		endif()
	endforeach()
endif()
