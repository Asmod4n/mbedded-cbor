	function(cbor_test_settings target)
		cmake_parse_arguments(PARSE_ARGV 1 arg "" "PCH_FROM" "PCH_HEADERS")
		if(NOT arg_PCH_HEADERS)
			set(arg_PCH_HEADERS ${CBOR_TEST_FULL_PCH_HEADERS})
		endif()
		target_link_libraries(${target} PRIVATE mbedded-cbor doctest::doctest)
		target_compile_options(${target} PRIVATE ${CBOR_WARNINGS} $<$<BOOL:${CBOR_HAS_WC2Y_EXTENSIONS}>:-Wno-c2y-extensions>)
		if(CBOR_HAS_FREFLECTION)
			target_compile_features(${target} PRIVATE cxx_std_26)
			target_compile_options(${target} PRIVATE -freflection)
		endif()
		target_compile_definitions(${target} PRIVATE DOCTEST_COUNTER=__COUNTER__)
		foreach(macro directory IN ZIP_LISTS CBOR_TEST_DATA_MACROS CBOR_TEST_DATA_DIRECTORIES)
			target_compile_definitions(${target} PRIVATE ${macro}="${directory}")
		endforeach()
		if(arg_PCH_FROM STREQUAL "${target}")
			target_precompile_headers(${target} PRIVATE ${arg_PCH_HEADERS})
		elseif(NOT arg_PCH_FROM STREQUAL "none")
			target_precompile_headers(${target} REUSE_FROM ${arg_PCH_FROM})
		endif()
	endfunction()

		function(cbor_test_stamp program)
			get_target_property(sources ${program} SOURCES)
			list(FILTER sources INCLUDE REGEX "\\.cpp$")
			set(data)
			foreach(source IN LISTS sources)
				file(READ ${PROJECT_SOURCE_DIR}/${source} text)
				foreach(macro directory IN ZIP_LISTS CBOR_TEST_DATA_MACROS CBOR_TEST_DATA_DIRECTORIES)
					if(text MATCHES "(^|[^A-Za-z0-9_])${macro}([^A-Za-z0-9_]|$)")
						file(GLOB_RECURSE files CONFIGURE_DEPENDS ${directory}/*)
						list(APPEND data ${files})
					endif()
				endforeach()
			endforeach()
			list(REMOVE_DUPLICATES data)
			set(stamp ${PROJECT_BINARY_DIR}/${program}.${CBOR_TEST_STAMP_NAME})
			add_custom_command(OUTPUT ${stamp}
				COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:${program}> "-DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}" "-DPATTERN=${CBOR_TEST_PATTERN}" -DLOG=${PROJECT_BINARY_DIR}/${program}.log -DSTAMP=${stamp} -P ${PROJECT_SOURCE_DIR}/cmake/test_stamp.cmake
				DEPENDS ${program} ${data} ${PROJECT_SOURCE_DIR}/cmake/test_stamp.cmake
				COMMENT "Running ${program}" JOB_POOL compile_job_pool VERBATIM)
			set(CBOR_TEST_STAMPS ${CBOR_TEST_STAMPS} ${stamp} PARENT_SCOPE)
		endfunction()
